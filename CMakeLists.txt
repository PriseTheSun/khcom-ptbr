cmake_minimum_required(VERSION 3.25)
if(CMAKE_HOST_APPLE AND NOT CMAKE_OSX_DEPLOYMENT_TARGET)
  set(CMAKE_OSX_DEPLOYMENT_TARGET "13.0" CACHE STRING "Proposed minimum macOS version" FORCE)
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>" CACHE STRING "Static MSVC runtime for standalone desktop packages")
project(khcom_ptbr_diagnostics VERSION 0.1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(KHCOM_BUILD_DESKTOP "Build the SDL desktop application" ON)
option(KHCOM_BUILD_DIAGNOSTIC "Build the SDL diagnostic harness" ON)
if(ANDROID)
  # Android will supply its own UI/audio/storage adapters. The portable model
  # and software GBA core remain available to an NDK consumer without SDL.
  set(KHCOM_BUILD_DESKTOP OFF CACHE BOOL "Build the SDL desktop application" FORCE)
  set(KHCOM_BUILD_DIAGNOSTIC OFF CACHE BOOL "Build the SDL diagnostic harness" FORCE)
endif()
include(cmake/Dependencies.cmake)
if(APPLE)
  # mGBA 0.10.x sets a directory-local 10.6 flag. Keep its source unmodified
  # while selecting the same deployment baseline as the containing program.
  target_compile_options(mgba PRIVATE "-mmacosx-version-min=${CMAKE_OSX_DEPLOYMENT_TARGET}")
endif()
add_library(khcom_core STATIC core/session.cpp core/sha256.cpp)
target_include_directories(khcom_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(khcom_core PUBLIC mgba)
target_compile_definitions(khcom_core PUBLIC KHCOM_DIAGNOSTIC_VERSION="${PROJECT_VERSION}")
if(MSVC)
  target_compile_options(khcom_core PRIVATE /W4)
  target_compile_options(khcom_core PUBLIC /utf-8)
else()
  target_compile_options(khcom_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(KHCOM_BUILD_DIAGNOSTIC)
  add_executable(khcom-diagnostic app/diagnostic.cpp platform/sdl_preview.cpp)
  target_link_libraries(khcom-diagnostic PRIVATE khcom_core SDL3::SDL3-static)
endif()

add_library(khcom_desktop_model STATIC app/desktop_model.cpp platform/storage_lock.cpp platform/atomic_file.cpp)
target_link_libraries(khcom_desktop_model PUBLIC khcom_core nlohmann_json::nlohmann_json miniz)
if(KHCOM_BUILD_DESKTOP)
  add_executable(khcom-desktop app/desktop.cpp platform/crt_shader.cpp)
  target_include_directories(khcom-desktop PRIVATE "${CMAKE_BINARY_DIR}/generated" "${sdl_SOURCE_DIR}/test")
  target_link_libraries(khcom-desktop PRIVATE khcom_desktop_model khcom_imgui)
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(khcom-core-tests tests/cpp/core_tests.cpp)
  target_link_libraries(khcom-core-tests PRIVATE khcom_core)
  target_compile_definitions(khcom-core-tests PRIVATE KHCOM_PINNED_CORE_VERSION="${KHCOM_PINNED_CORE_VERSION}")
  add_test(NAME core-contracts COMMAND khcom-core-tests)
  if(KHCOM_BUILD_DIAGNOSTIC)
    add_test(NAME diagnostic-help COMMAND khcom-diagnostic --help)
  endif()
  add_executable(khcom-desktop-tests tests/cpp/desktop_tests.cpp)
  target_link_libraries(khcom-desktop-tests PRIVATE khcom_desktop_model)
  add_test(NAME desktop-contracts COMMAND khcom-desktop-tests)
  add_executable(khcom-storage-integration tests/cpp/private_integration.cpp)
  target_link_libraries(khcom-storage-integration PRIVATE khcom_desktop_model)
  if(KHCOM_BUILD_DESKTOP)
    add_test(NAME desktop-help COMMAND khcom-desktop --help)
  endif()
endif()

# Packaging is owned by the release workflow; no game payload is embedded.
